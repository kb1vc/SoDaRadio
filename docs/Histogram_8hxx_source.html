50 Histogram(unsigned int _num_buckets,
51 double _min, double _max) {
52 num_buckets = _num_buckets;
53 min = _min;
54 max = _max;
55 table = new unsigned int [num_buckets];
56 recip_bucket_interval = ((double) num_buckets) / (max - min);
57 for(unsigned int i = 0; i < num_buckets; i++) table[i] = 0;
58 }
59
60 void updateTable(double v) {
61 int idx = lround((v - min) * recip_bucket_interval);
62 if(v > max) table[num_buckets - 1]++;
63 else if(v < min) table[0]++;
64 else table[idx]++;
65 }
66
67 void writeTable(const std::string & fname) {
68 std::ofstream of(fname.c_str());
69
70 double bucket_interval = 1.0 / recip_bucket_interval;
71 double half_int = 0.5 * bucket_interval;
72
73 for(unsigned int i = 0; i < num_buckets; i++) {
74 double mid = min + half_int + ((double) i) * bucket_interval;
75 if(table[i] != 0) of << SoDa::Format("%0 %1 %2\n")
76 .addI(i)
77 .addF(mid)
78 .addI(table[i]);
79 }
80
81 of.close();
82 }

84 protected:

87 unsigned int num_buckets;
88 double min, max;
