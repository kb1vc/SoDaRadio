119 // scan the buffer until we get to the newline.

123 ch = inbuf.front(); inbuf.pop();

131 }
132 // if we get here, we ran out of buffer space.

134 // consume the queue up to the newline.
135 for(ch = '\000'; (ch != '\n') && !inbuf.empty(); ) {
